  // Paste the deployed Apps Script /exec URL here

  const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbwTyNGC9SvJWqnstceylgb3O9mD9G2UysGsmJTbt4q47BY7xtSyq6m5G78tz1XFzMiNIw/exec';

  const SECRET = 'wishpond2026';

  let data = { months: [] };

  function toast(msg, type) {

    const el = document.getElementById('toast');

    el.textContent = msg;

    el.className = 'toast show' + (type ? ' ' + type : '');

    clearTimeout(el._t);

    el._t = setTimeout(() => el.classList.remove('show'), 3000);

  }

  function esc(s) {

    return String(s || '')

      .replace(/&/g, '&amp;')

      .replace(/</g, '&lt;')

      .replace(/>/g, '&gt;');

  }

  function mrrNum(s) {

    const n = parseFloat(String(s || '').replace(/[^0-9.]/g, ''));

    return isNaN(n) ? 0 : n;

  }

  function fmt(n) {

    return n > 0 ? '$' + n.toLocaleString() : '—';

  }

  // LOAD FROM SHEET

  function pullFromSheet() {

    if (SCRIPT_URL.indexOf('PASTE') > -1) {

      toast('Add your Apps Script URL to the code first', 'err');

      return;

    }

    const btn = document.getElementById('pullBtn');

    btn.disabled = true;

    btn.textContent = 'Refreshing…';

    const cb = 'wp_pull_' + Date.now();

    let done = false;

    window[cb] = function (res) {

      done = true;

      delete window[cb];

      const script = document.getElementById(cb);

      if (script) script.remove();

      btn.disabled = false;

      btn.textContent = '↓ Refresh from sheet';

      if (res && res.ok && res.data) {

        data = res.data;

        render();

        toast('Loaded from sheet ✓', 'ok');

      } else {

        toast(

          'Could not load: ' +

          (res && res.error ? res.error : 'error'),

          'err'

        );

      }

    };

    setTimeout(() => {

      if (!done && window[cb]) {

        delete window[cb];

        btn.disabled = false;

        btn.textContent = '↓ Refresh from sheet';

        toast('Timed out', 'err');

      }

    }, 10000);

    const script = document.createElement('script');

    script.id = cb;

    script.src =

      SCRIPT_URL +

      '?callback=' +

      cb +

      '&t=' +

      Date.now();

    document.head.appendChild(script);

  }

  // SAVE TO SHEET

  function saveToSheet() {

    if (SCRIPT_URL.indexOf('PASTE') > -1) {

      toast('Add your Apps Script URL to the code first', 'err');

      return;

    }

    const btn = document.getElementById('saveBtn');

    btn.disabled = true;

    btn.innerHTML = '<span class="spinner"></span>Saving…';

    fetch(SCRIPT_URL, {

      method: 'POST',

      mode: 'no-cors',

      headers: {

        'Content-Type': 'text/plain;charset=utf-8'

      },

      body: JSON.stringify({

        secret: SECRET,

        data: data

      })

    })

      .then(() => {

        const cb = 'wp_verify_' + Date.now();

        let done = false;

        window[cb] = function (res) {

          done = true;

          delete window[cb];

          const script = document.getElementById(cb);

          if (script) script.remove();

          btn.disabled = false;

          btn.textContent = '↑ Save to sheet';

          if (res && res.ok) {

            document.getElementById('savetag').textContent =

              'saved ✓';

            setTimeout(() => {

              document.getElementById('savetag').textContent = '';

            }, 3000);

          } else {

            toast('Save may have failed', 'err');

          }

        };

        setTimeout(() => {

          if (!done && window[cb]) {

            delete window[cb];

            btn.disabled = false;

            btn.textContent = '↑ Save to sheet';

            document.getElementById('savetag').textContent =

              'saved ✓';

            setTimeout(() => {

              document.getElementById('savetag').textContent = '';

            }, 3000);

          }

        }, 8000);

        const verifyScript = document.createElement('script');

        verifyScript.id = cb;

        verifyScript.src =

          SCRIPT_URL +

          '?callback=' +

          cb +

          '&t=' +

          Date.now();

        setTimeout(() => {

          document.head.appendChild(verifyScript);

        }, 1500);

      })

      .catch((err) => {

        btn.disabled = false;

        btn.textContent = '↑ Save to sheet';

        toast('Could not reach sheet: ' + err.message, 'err');

      });

  }

  // RENDER

  function render() {

    let totalClients = 0;

    let totalMrr = 0;

    (data.months || []).forEach((month) => {

      (month.clients || []).forEach((client) => {

        totalClients++;

        totalMrr += mrrNum(client.mrr);

      });

    });

    document.getElementById('summary').innerHTML = `

      <div class="stat churn">

        <div class="stat-label">Total churned</div>

        <div class="stat-num">${totalClients}</div>

        <div class="stat-sub">across ${(data.months || []).length} month(s)</div>

      </div>

      <div class="stat mrr">

        <div class="stat-label">MRR lost</div>

        <div class="stat-num">${fmt(totalMrr)}</div>

        <div class="stat-sub">total monthly recurring</div>

      </div>

      <div class="stat clients">

        <div class="stat-label">Avg MRR / client</div>

        <div class="stat-num">${totalClients ? fmt(Math.round(totalMrr / totalClients)) : '—'}</div>

        <div class="stat-sub">per churned account</div>

      </div>

      <div class="stat months">

        <div class="stat-label">Months tracked</div>

        <div class="stat-num">${(data.months || []).length}</div>

        <div class="stat-sub">sections in sheet</div>

      </div>

    `;

    const content = document.getElementById('content');

    content.innerHTML = '';

    (data.months || []).forEach((month, monthIndex) => {

      let monthMrr = 0;

      (month.clients || []).forEach((client) => {

        monthMrr += mrrNum(client.mrr);

      });

      const block = document.createElement('div');

      block.className = 'month-block';

      block.innerHTML = `

        <div class="month-head">

          <span

            class="month-name"

            contenteditable="true"

            data-mi="${monthIndex}"

            oninput="editMonth(${monthIndex}, this.textContent)"

          >${esc(month.name)}</span>

          <span class="month-count">

            ${(month.clients || []).length} clients

          </span>

          <span class="month-mrr">

            ${fmt(monthMrr)} MRR lost

          </span>

        </div>

      `;

      const wrap = document.createElement('div');

      wrap.className = 'tbl-wrap';

      const table = document.createElement('table');

      table.className = 'tbl';

      table.innerHTML = `

        <thead>

          <tr>

            <th>Client Name</th>

            <th>AM</th>

            <th>CSM</th>

            <th>MRR</th>

            <th>Brand</th>

            <th>Churn Date</th>

            <th>Reason for Churn</th>

            <th></th>

          </tr>

        </thead>

      `;

      const body = document.createElement('tbody');

      (month.clients || []).forEach((client, clientIndex) => {

        const row = document.createElement('tr');

        row.innerHTML = `

          <td data-label="Client">

            <span class="cell" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'client',this.textContent)"

            >${esc(client.client)}</span>

          </td>

          <td data-label="AM">

            <span class="cell" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'am',this.textContent)"

            >${esc(client.am)}</span>

          </td>

          <td data-label="CSM">

            <span class="cell" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'csm',this.textContent)"

            >${esc(client.csm)}</span>

          </td>

          <td data-label="MRR">

            <span class="cell mrr" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'mrr',this.textContent)"

            >${esc(client.mrr)}</span>

          </td>

          <td data-label="Brand">

            <span class="cell" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'brand',this.textContent)"

            >${esc(client.brand)}</span>

          </td>

          <td data-label="Churn Date">

            <span class="cell" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'churnDate',this.textContent)"

            >${esc(client.churnDate)}</span>

          </td>

          <td data-label="Reason">

            <span class="cell" contenteditable="true"

              oninput="ed(${monthIndex},${clientIndex},'reason',this.textContent)"

            >${esc(client.reason)}</span>

          </td>

          <td>

            <button

              class="del"

              onclick="delClient(${monthIndex},${clientIndex})"

              title="Remove"

            >×</button>

          </td>

        `;

        body.appendChild(row);

      });

      table.appendChild(body);

      wrap.appendChild(table);

      block.appendChild(wrap);

      const addButton = document.createElement('button');

      addButton.className = 'add-client';

      addButton.textContent = '+ add client to ' + month.name;

      addButton.onclick = () => addClient(monthIndex);

      block.appendChild(addButton);

      content.appendChild(block);

    });

    if (!(data.months || []).length) {

      content.innerHTML =

        '<div class="loading">No data yet. Click "+ add month" to start, or edit your Google Sheet and Refresh.</div>';

    }

  }

  // EDIT HANDLERS

  function ed(monthIndex, clientIndex, field, value) {

    data.months[monthIndex].clients[clientIndex][field] =

      value.trim();

  }

  function editMonth(monthIndex, value) {

    data.months[monthIndex].name = value.trim();

  }

  function addClient(monthIndex) {

    data.months[monthIndex].clients.push({

      client: '',

      am: '',

      csm: '',

      mrr: '',

      brand: '',

      churnDate: '',

      reason: ''

    });

    render();

  }

  function delClient(monthIndex, clientIndex) {

    if (!confirm('Remove this client?')) return;

    data.months[monthIndex].clients.splice(clientIndex, 1);

    render();

  }

  function addMonth() {

    const name = prompt('Month name? (e.g. August)');

    if (!name) return;

    data.months.push({

      name: name.trim(),

      clients: []

    });

    render();

  }

  // BOOT

  render();

  pullFromSheet();
